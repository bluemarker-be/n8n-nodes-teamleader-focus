### Task 2: Meeting execute logic — build location object

**Context:** Both code paths read flat fields from `additionalFields` / `updateFields`, then call `assignDefined(body, fields)` which copies all remaining keys to the body. We need to intercept the `location_*` keys, build the nested object, and delete them before `assignDefined` runs. Same pattern as the existing `customer_type`/`customer_id` nesting.

- [ ] **Step 1: Add a helper function `buildLocationObject`**

Add this function near the other helper functions in `TeamleaderFocus.node.ts` (e.g. around line 3270, before `buildInvoiceBody`):

```typescript
function buildLocationObject(fields: IDataObject): IDataObject | undefined {
	const locationType = fields.location_type as string | undefined;
	if (!locationType) return undefined;

	const address: IDataObject = {};
	if (fields.location_line_1) address.line_1 = fields.location_line_1;
	if (fields.location_postal_code) address.postal_code = fields.location_postal_code;
	if (fields.location_city) address.city = fields.location_city;
	if (fields.location_country) address.country = fields.location_country;

	let location: IDataObject;
	switch (locationType) {
		case 'virtual':
			location = { type: 'virtual' };
			break;
		case 'contact':
		case 'company':
			location = { type: locationType, id: fields.location_id, address };
			break;
		case 'customLocation':
			location = { type: 'customLocation', address };
			break;
		case 'calendarResource':
			location = { type: 'calendarResource', id: fields.location_id };
			break;
		default:
			location = { type: locationType };
	}
	return location;
}

function deleteLocationFields(fields: IDataObject): void {
	delete fields.location_type;
	delete fields.location_id;
	delete fields.location_line_1;
	delete fields.location_postal_code;
	delete fields.location_city;
	delete fields.location_country;
}
```
